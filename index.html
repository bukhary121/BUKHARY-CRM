<script>
async function getLeads() {
  const city = document.getElementById("city").value;
  const type = document.getElementById("type").value;
  const limit = document.getElementById("limit").value;

  const query = `
  [out:json][timeout:25];
  area["name"="${city}"]->.searchArea;
  (
    node["amenity"="${type}"](area.searchArea);
    node["shop"](area.searchArea);
  );
  out ${limit};
  `;

  const url = "https://overpass-api.de/api/interpreter";

  const res = await fetch(url, {
    method: "POST",
    body: query
  });

  const json = await res.json();

  const table = document.getElementById("data");
  table.innerHTML = "";

  json.elements.forEach(el => {
    const name = el.tags?.name || "غير معروف";
    const map = `https://www.openstreetmap.org/node/${el.id}`;

    table.innerHTML += `
      <tr>
        <td>${name}</td>
        <td>${city}</td>
        <td>${type}</td>
        <td><a href="${map}" target="_blank">فتح</a></td>
        <td><button onclick="send('${name}')">إرسال عرض</button></td>
      </tr>
    `;
  });
}

function send(name) {
  const message = `مرحبا 👋
لاحظنا نشاطكم (${name}) يحتاج تطوير رقمي

نقدر نساعدكم بـ:
✔️ تصميم متجر
✔️ إدارة سوشيال ميديا
✔️ زيادة العملاء

هل مناسب نتواصل؟`;

  navigator.clipboard.writeText(message);

  alert("تم نسخ رسالة العرض 👍");
}
